async function generateCode() { const apiKey = document.getElementById("apiKeyInput").value; // आपकी API Key वाला बॉक्स const userPrompt = document.getElementById("promptInput").value; // यूजर का सवाल वाला बॉक्स const previewBox = document.getElementById("previewBox"); // नीचे वाला सफेद बॉक्स previewBox.innerHTML = "AI सोच रहा है... कृपया प्रतीक्षा करें..."; const url = `https://googleapis.com{apiKey}`; const response = await fetch(url, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ contents: [{ parts: [{ text: userPrompt }] }] }) }); const data = await response.json(); // AI का जवाब निकालना if (data.candidates) { const aiReply = data.candidates[0].content.parts[0].text; previewBox.innerText = aiReply; // नीचे सफेद बॉक्स में जवाब दिखाना } else { previewBox.innerText = "Error: कृपया अपनी API Key चेक करें!"; } }
